# 009 | 梯度：函数增长最快的方向

> 🎯 **一句话核心**：梯度是多元函数的"导数"，指向函数增长最快的方向——是神经网络训练（梯度下降）的基础。

---

## 🤔 从一个问题开始

### 问题：如何在雾中下山？

你在山上，雾很大，看不到山脚。如何找到下山的路？

**答案**：每次感受脚下哪个方向最陡，往那个方向走一步。

这就是**梯度下降**的核心思想！

---

### 从导数到梯度

**一元函数**：导数 f'(x) 表示：
- 变化率
- 切线斜率
- x 增加时 f 的变化

**多元函数**：梯度 ∇f(x) 表示：
- 每个方向的偏导数组成的向量
- 指向函数增长最快的方向
- 长度表示增长的"陡峭"程度

---

### 梯度的定义

对于函数 f(x₁, x₂, ..., xₙ)：

```
梯度 ∇f = [∂f/∂x₁, ∂f/∂x₂, ..., ∂f/∂xₙ]

其中 ∂f/∂xᵢ 是 f 对 xᵢ 的偏导数
```

---

## 📖 核心概念详解

### 1. 偏导数

**定义**：固定其他变量，只对其中一个变量求导。

**例子**：f(x, y) = x² + 2xy + y²

```
∂f/∂x = 2x + 2y  （把 y 当常数）
∂f/∂y = 2x + 2y  （把 x 当常数）

梯度 ∇f = [2x + 2y, 2x + 2y]
```

---

### 2. 梯度的几何意义

**关键性质**：梯度指向函数增长最快的方向！

```
在点 (x₀, y₀)：
- 梯度方向：函数增长最快的方向
- 负梯度方向：函数下降最快的方向
- 梯度长度：增长的速率
```

**等高线视角**：
- 梯度垂直于等高线
- 指向等高线密集的方向（陡峭）

---

### 3. 梯度下降法

**目标**：最小化函数 f(x)

**算法**：
```
初始化 x
repeat:
    x = x - α × ∇f(x)
until 收敛

其中 α 是学习率
```

**直观**：每次往最陡的下降方向走一步。

---

### 4. 学习率的选择

| 学习率 α | 效果 |
|----------|------|
| 太小 | 收敛慢，训练时间长 |
| 太大 | 可能发散，在最优解附近震荡 |
| 合适 | 稳定收敛 |

**自适应学习率**：Adam、RMSprop 等算法动态调整学习率。

---

### 5. 梯度的计算

#### 手动求导

适用于简单函数，精确但繁琐。

#### 数值梯度

```
∂f/∂x ≈ (f(x+h) - f(x)) / h

简单但计算量大，有数值误差
```

#### 自动微分（深度学习框架使用）

```
PyTorch、TensorFlow 使用反向模式自动微分
高效精确地计算梯度
```

## 💻 代码实现

```python
import numpy as np
import matplotlib.pyplot as plt

# ==================== 偏导数计算 ====================

print("="*50)
print("偏导数与梯度计算")
print("="*50)

def f(x, y):
    """目标函数：f(x, y) = x² + 2xy + y²"""
    return x**2 + 2*x*y + y**2

def gradient_f(x, y):
    """梯度：∇f = [2x+2y, 2x+2y]"""
    df_dx = 2*x + 2*y
    df_dy = 2*x + 2*y
    return np.array([df_dx, df_dy])

# 测试
x, y = 1, 2
print(f"\n函数 f(x,y) = x² + 2xy + y²")
print(f"\n在点 ({x}, {y})：")
print(f"  f({x}, {y}) = {f(x, y)}")
print(f"  ∂f/∂x = {2*x + 2*y}")
print(f"  ∂f/∂y = {2*x + 2*y}")
print(f"  梯度 ∇f = {gradient_f(x, y)}")

# ==================== 数值梯度验证 ====================

print("\n" + "="*50)
print("数值梯度 vs 解析梯度")
print("="*50)

def numerical_gradient(f, x, h=1e-5):
    """数值梯度计算"""
    grad = np.zeros_like(x)
    for i in range(len(x)):
        x_plus = x.copy()
        x_minus = x.copy()
        x_plus[i] += h
        x_minus[i] -= h
        grad[i] = (f(x_plus) - f(x_minus)) / (2*h)
    return grad

# 测试函数
def f2(x):
    return x[0]**2 + 2*x[0]*x[1] + x[1]**2

point = np.array([1.0, 2.0])

grad_analytical = gradient_f(point[0], point[1])
grad_numerical = numerical_gradient(f2, point)

print(f"\n点：{point}")
print(f"解析梯度：{grad_analytical}")
print(f"数值梯度：{grad_numerical}")
print(f"误差：{np.max(np.abs(grad_analytical - grad_numerical)):.2e}")

# ==================== 梯度下降可视化 ====================

print("\n" + "="*50)
print("梯度下降法可视化")
print("="*50)

def gradient_descent(f, grad_f, x0, learning_rate=0.1, n_iter=50):
    """
    梯度下降优化
    """
    x = x0.copy()
    history = [x.copy()]
    
    for i in range(n_iter):
        g = grad_f(x[0], x[1])
        x = x - learning_rate * g
        history.append(x.copy())
    
    return np.array(history)

# 创建网格
x = np.linspace(-3, 3, 100)
y = np.linspace(-3, 3, 100)
X, Y = np.meshgrid(x, y)
Z = X**2 + 2*X*Y + Y**2

# 运行梯度下降
x0 = np.array([2.5, 2.5])
trajectory = gradient_descent(f, gradient_f, x0, learning_rate=0.1, n_iter=30)

# 可视化
fig = plt.figure(figsize=(14, 5))

# 3D 表面
ax1 = fig.add_subplot(131, projection='3d')
ax1.plot_surface(X, Y, Z, alpha=0.7, cmap='viridis')
ax1.plot(trajectory[:, 0], trajectory[:, 1], 
         f(trajectory[:, 0], trajectory[:, 1]), 
         'r-o', linewidth=2, markersize=3, label='梯度下降路径')
ax1.set_xlabel('x')
ax1.set_ylabel('y')
ax1.set_zlabel('f(x,y)')
ax1.set_title('3D 表面与优化路径')
ax1.legend()

# 等高线
ax2 = fig.add_subplot(132)
contour = ax2.contourf(X, Y, Z, levels=50, cmap='viridis')
ax2.plot(trajectory[:, 0], trajectory[:, 1], 'w-o', 
         linewidth=2, markersize=3, label='梯度下降路径')
ax2.set_xlabel('x')
ax2.set_ylabel('y')
ax2.set_title('等高线与优化路径')
ax2.legend()
plt.colorbar(contour, ax=ax2)

# 函数值变化
ax3 = fig.add_subplot(133)
f_values = f(trajectory[:, 0], trajectory[:, 1])
ax3.plot(f_values, 'b-', linewidth=2)
ax3.set_xlabel('迭代次数')
ax3.set_ylabel('f(x,y)')
ax3.set_title('函数值收敛曲线')
ax3.grid(True, alpha=0.3)
ax3.set_yscale('log')

plt.tight_layout()
plt.savefig('/Users/cloudyan/.openclaw/workspace/memory/ai-math-tutorial/images/009-gradient-descent.png', dpi=150)
print("\n梯度下降可视化已保存：images/009-gradient-descent.png")

# ==================== 学习率的影响 ====================

print("\n" + "="*50)
print("学习率对梯度下降的影响")
print("="*50)

learning_rates = [0.01, 0.05, 0.1, 0.2, 0.5]

plt.figure(figsize=(12, 8))

for i, lr in enumerate(learning_rates):
    trajectory = gradient_descent(f, gradient_f, x0, learning_rate=lr, n_iter=20)
    f_values = f(trajectory[:, 0], trajectory[:, 1])
    
    plt.subplot(2, 3, i+1)
    plt.plot(f_values, 'b-', linewidth=2)
    plt.title(f'学习率 α = {lr}')
    plt.xlabel('迭代次数')

# ==================== 梯度指向增长最快的方向 ====================

print("\n" + "="*50)
print("验证：梯度指向增长最快的方向")
print("="*50)

# 在某个点，比较不同方向的方向导数
point = np.array([1.0, 1.0])
grad = gradient_f(point[0], point[1])

# 归一化梯度
grad_unit = grad / np.linalg.norm(grad)

print(f"\n测试点：{point}")
print(f"梯度：{grad}")
print(f"梯度方向单位向量：{grad_unit}")

# 测试不同方向的方向导数
angles = np.linspace(0, 2*np.pi, 36)
directional_derivatives = []

for angle in angles:
    # 方向向量
    direction = np.array([np.cos(angle), np.sin(angle)])
    
    # 方向导数 = 梯度·方向
    dir_deriv = np.dot(grad, direction)
    directional_derivatives.append(dir_deriv)

# 找到最大方向导数的方向
max_idx = np.argmax(directional_derivatives)
max_angle = angles[max_idx]
max_direction = np.array([np.cos(max_angle), np.sin(max_angle)])

print(f"\n最大方向导数：{directional_derivatives[max_idx]:.4f}")
print(f"对应角度：{np.degrees(max_angle):.1f}°")
print(f"对应方向：{max_direction}")
print(f"梯度方向：{grad_unit}")
print(f"\n（两者一致 ✓）")

# 可视化方向导数
plt.figure(figsize=(8, 4))

plt.subplot(1, 2, 1)
plt.plot(np.degrees(angles), directional_derivatives, 'b-o')
plt.xlabel('方向角度 (度)')
plt.ylabel('方向导数')
plt.title('不同方向的方向导数')
plt.grid(True, alpha=0.3)
plt.axvline(np.degrees(max_angle), color='r', linestyle='--', label='最大方向')
plt.legend()

plt.subplot(1, 2, 2)
# 绘制方向向量和梯度
theta = np.linspace(0, 2*np.pi, 100)
plt.plot(np.cos(theta), np.sin(theta), 'gray', alpha=0.3)  # 单位圆
plt.quiver(0, 0, grad_unit[0], grad_unit[1], 
           angles='xy', scale_units='xy', scale=1, 
           color='r', width=0.015, label='梯度方向')
plt.quiver(0, 0, max_direction[0], max_direction[1],
           angles='xy', scale_units='xy', scale=1,
           color='b', width=0.01, label='最大增长方向', alpha=0.7)
plt.xlabel('x')
plt.ylabel('y')
plt.title('梯度方向 = 最大增长方向')
plt.legend()
plt.axis('equal')
plt.grid(True, alpha=0.3)

plt.tight_layout()

print("梯度方向验证图已保存：images/009-gradient-direction.png")

# ==================== 神经网络中的梯度 ====================

print("\n" + "="*50)
print("应用：简单神经网络的梯度")
print("="*50)

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def sigmoid_derivative(x):
    s = sigmoid(x)
    return s * (1 - s)

def mse_loss(y_pred, y_true):
    return np.mean((y_pred - y_true)**2)

def mse_loss_derivative(y_pred, y_true):
    return 2 * (y_pred - y_true) / len(y_true)

# 简单神经网络：输入 → 权重 → 输出
np.random.seed(42)
X = np.random.randn(100, 2)  # 100 个样本，2 个特征
y = (X[:, 0] + X[:, 1] > 0).astype(float)  # 简单分类任务

# 初始化权重
W = np.random.randn(2, 1) * 0.1
b = np.zeros(1)

# 前向传播
def forward(X, W, b):
    z = X @ W + b
    a = sigmoid(z)
    return a

# 计算梯度
def compute_gradients(X, y, W, b):
    # 前向
    z = X @ W + b
    a = sigmoid(z)
    
    # 反向
    dz = mse_loss_derivative(a, y) * sigmoid_derivative(z)
    dW = X.T @ dz
    db = np.sum(dz)
    
    return dW, db

# 训练
learning_rate = 0.5
for epoch in range(1000):
    dW, db = compute_gradients(X, y, W, b)
    W = W - learning_rate * dW
    b = b - learning_rate * db
    
    if epoch % 100 == 0:
        y_pred = forward(X, W, b)
        loss = mse_loss(y_pred, y)
        print(f"Epoch {epoch}: Loss = {loss:.4f}")

print(f"\n最终权重：W = {W.flatten()}")
print(f"最终偏置：b = {b}")

## 🔗 与 AI 的联系

### 1. 神经网络训练

反向传播 = 计算损失函数对权重的梯度

```
1. 前向传播：计算输出
2. 计算损失
3. 反向传播：计算梯度
4. 梯度下降：更新权重
```

---

### 2. 优化算法

- **SGD**：随机梯度下降
- **Momentum**：加入动量
- **Adam**：自适应学习率

---

### 3. 梯度问题

- **梯度消失**：深层网络梯度趋近 0
- **梯度爆炸**：梯度变得极大
- **解决**：ReLU、BatchNorm、残差连接

## 🧠 常见误区

### ❌ 误区 1：梯度指向最大值

**纠正**：梯度指向局部增长最快的方向，不一定是全局最大值。

可能陷入局部最优。

---

### ❌ 误区 2：学习率越大越好

**纠正**：学习率太大会导致发散！

需要仔细调参或用自适应方法。

---

### ❌ 误区 3：梯度为 0 就是最优解

**纠正**：梯度为 0 可能是：
- 局部最小值 ✓
- 局部最大值 ✗
- 鞍点（某些方向上升，某些下降）✗

def gradient_descent_momentum(grad_f, x0, lr=0.01, momentum=0.9, n_iter=100):
    x = x0.copy()
    v = np.zeros_like(x)  # 速度
    
    for _ in range(n_iter):
        g = grad_f(x)
        v = momentum * v + lr * g
        x = x - v
    
    return x
```
